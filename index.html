    const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


    let matches = [];

    let updates = [];


    // --- Tab Switcher ---

    function switchTab(tab) {

      ['matches', 'updates'].forEach(t => {

        document.getElementById(`sec-${t}`).classList.add('hidden');

        document.getElementById(`tab-${t}`).classList.remove('active-tab');

      });

      document.getElementById(`sec-${tab}`).classList.remove('hidden');

      document.getElementById(`tab-${tab}`).classList.add('active-tab');

    }


    // --- Fetch Data ---

    async function fetchData() {

      const { data: matchData } = await supabaseClient.from('matches').select('*').order('id', { ascending: true });

      const { data: updateData } = await supabaseClient.from('updates').select('*').order('created_at', { ascending: false });

      

      if(matchData) matches = matchData;

      if(updateData) updates = updateData;

      

      renderAll();

    }


    // --- Subscribe to Live Database Changes ---

    function setupRealtimeSubscriptions() {

      supabaseClient

        .channel('public-schema-changes')

        .on('postgres_changes', { event: '*', schema: 'public', table: 'matches' }, () => fetchData())

        .on('postgres_changes', { event: '*', schema: 'public', table: 'updates' }, () => fetchData())

        .subscribe();

    }


    // --- Render UI ---

    function renderAll() {

      // Matches

      const matchesEl = document.getElementById('matches-list');

      const matchSelect = document.getElementById('admin-match-select');

      matchesEl.innerHTML = '';

      matchSelect.innerHTML = '';


      matches.forEach(m => {

        const isLive = m.status.includes('LIVE');

        matchesEl.innerHTML += `

          <div class="bg-white p-4 rounded-lg shadow border-l-4 ${isLive ? 'border-red-500' : 'border-emerald-600'} flex justify-between items-center">

            <div class="space-y-1 w-2/3">

              <div class="flex justify-between font-bold text-sm text-gray-800">

                <span>${m.home_team}</span> <span>${m.home_score}</span>

              </div>

              <div class="flex justify-between font-bold text-sm text-gray-800">

                <span>${m.away_team}</span> <span>${m.away_score}</span>

              </div>

            </div>

            <div class="text-right border-l pl-3">

              <span class="text-[10px] font-bold px-2 py-0.5 rounded ${isLive ? 'bg-red-100 text-red-600 animate-pulse' : 'bg-gray-100 text-gray-600'}">

                ${m.status}

              </span>

              <p class="text-xs text-gray-400 mt-1">${m.match_time || ''}</p>

            </div>

          </div>`;


        matchSelect.innerHTML += `<option value="${m.id}">${m.home_team} vs ${m.away_team}</option>`;

      });


      // Updates

      const updatesEl = document.getElementById('updates-list');

      updatesEl.innerHTML = updates.map(u => {

        const timeStr = new Date(u.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

        return `

          <div class="bg-white p-3 rounded-lg shadow-sm border-l-2 border-blue-500">

            <p class="text-xs text-gray-400 font-bold mb-1">${timeStr}</p>

            <p class="text-sm text-gray-700">${u.update_text}</p>

          </div>`;

      }).join('');

    }


    // --- Admin Handlers ---

    async function updateScore() {

      const matchId = parseInt(document.getElementById('admin-match-select').value);

      const hScore = parseInt(document.getElementById('home-score').value);

      const aScore = parseInt(document.getElementById('away-score').value);

      const status = document.getElementById('match-status').value;


      if(isNaN(hScore) || isNaN(aScore)) return alert('Enter valid scores');


      await supabaseClient

        .from('matches')

        .update({ home_score: hScore, away_score: aScore, status: status })

        .eq('id', matchId);


      alert('Score updated live!');

    }


    async function addUpdate() {

      const text = document.getElementById('admin-update-text').value;

      if(!text) return;


      await supabaseClient.from('updates').insert([{ update_text: text }]);

      document.getElementById('admin-update-text').value = '';

    }


    // --- Modal & Auth UI ---

    function toggleAdminModal() {

      document.getElementById('login-modal').classList.toggle('hidden');

      document.getElementById('login-modal').classList.toggle('flex');

    }


    function loginAdmin() {

      if(document.getElementById('admin-pass').value === 'admin123') {

        document.getElementById('sec-admin-panel').classList.remove('hidden');

        document.getElementById('admin-btn-text').innerText = 'Admin Active';

      } else {

        alert('Incorrect password!');

      }

    }


    function logoutAdmin() {

      document.getElementById('sec-admin-panel').classList.add('hidden');

      document.getElementById('admin-btn-text').innerText = 'Admin';

    }
